# Unofficial non-commercial MODU-C dongle adaptation; see root LICENSE/NOTICE.
# These are evaluated against the REAL generated Kconfig during a ZMK build.
if(CONFIG_SHIELD_MODU_DONGLE)
  if(NOT CONFIG_ZMK_SPLIT_ROLE_CENTRAL OR NOT CONFIG_ZMK_USB)
    message(FATAL_ERROR "MODU dongle must be the USB-capable split central")
  endif()
  if(NOT CONFIG_ZMK_SPLIT_BLE_CENTRAL_PERIPHERALS EQUAL 2)
    message(FATAL_ERROR "MODU dongle must accept exactly two peripherals")
  endif()
  if(NOT CONFIG_ZMK_POINTING OR NOT CONFIG_ZMK_DISPLAY)
    message(FATAL_ERROR "MODU dongle pointing/OLED configuration was not loaded")
  endif()
  message(STATUS "MODU dongle: central, two peers, pointing and OLED enabled")
endif()

if(CONFIG_SHIELD_MODU_PERIPHERAL_LEFT OR CONFIG_SHIELD_MODU_PERIPHERAL_RIGHT)
  if(CONFIG_ZMK_SPLIT_ROLE_CENTRAL OR CONFIG_ZMK_USB)
    message(FATAL_ERROR "MODU left/right dongle builds must be peripherals without host USB HID")
  endif()
  if(NOT CONFIG_ZMK_POINTING OR NOT CONFIG_PMW3610_ALT OR NOT CONFIG_MODU_ALT_THUMB_KSCAN)
    message(FATAL_ERROR "MODU peripheral trackball/orientation configuration was not loaded")
  endif()
  message(STATUS "MODU half: peripheral, original PMW3610 and thumb scan enabled")
endif()
