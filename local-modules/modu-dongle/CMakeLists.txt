# Standard Zephyr module library; also works while find_package(Zephyr) loads modules.
if(NOT CONFIG_SHIELD_MODU_DONGLE AND NOT CONFIG_SHIELD_MODU_DONGLE_OLED
   AND NOT CONFIG_MODU_PERIPHERAL_STATUS_LED
   AND NOT CONFIG_MODU_BATTERY_TELEMETRY_PERIPHERAL
   AND NOT CONFIG_MODU_BATTERY_TELEMETRY_CENTRAL)
  return()
endif()
zephyr_library_named(modu_dongle_runtime)
zephyr_library_include_directories(${APPLICATION_SOURCE_DIR}/include)
if(CONFIG_SHIELD_MODU_DONGLE OR CONFIG_SHIELD_MODU_DONGLE_OLED)
  set(_modu_keymap "${ZMK_CONFIG}/modu.keymap")
  if(NOT EXISTS "${_modu_keymap}")
    message(FATAL_ERROR "The canonical custom keymap is missing: ${_modu_keymap}")
  endif()
  file(SHA256 "${_modu_keymap}" _modu_keymap_sha)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${_modu_keymap}")
  message(STATUS "MODU-C v3 custom keymap SHA256: ${_modu_keymap_sha}")
  zephyr_library_compile_definitions(MODU_CONFIG_KEYMAP_SHA256="${_modu_keymap_sha}")
  zephyr_library_sources(${CMAKE_CURRENT_LIST_DIR}/src/keymap_guard.c)
endif()
if(CONFIG_MODU_PERIPHERAL_STATUS_LED)
  if(CONFIG_MODU_LED_BREATH)
    message(FATAL_ERROR "Disable MODU_LED_BREATH: two LED drivers must not drive the same PWM pins")
  endif()
  zephyr_library_sources(${CMAKE_CURRENT_LIST_DIR}/src/peripheral_status_led.c)
endif()
if(CONFIG_MODU_DONGLE_STATUS_SCREEN)
  if(NOT CONFIG_ZMK_DISPLAY_STATUS_SCREEN_CUSTOM)
    message(FATAL_ERROR "MODU-C OLED requires ZMK_DISPLAY_STATUS_SCREEN_CUSTOM")
  endif()
  include(${CMAKE_CURRENT_LIST_DIR}/original_display.cmake)
endif()

if(CONFIG_MODU_BATTERY_TELEMETRY_PERIPHERAL)
  zephyr_library_sources(${CMAKE_CURRENT_LIST_DIR}/src/battery_telemetry_peripheral.c)
endif()
if(CONFIG_MODU_BATTERY_TELEMETRY_CENTRAL)
  zephyr_library_sources(${CMAKE_CURRENT_LIST_DIR}/src/battery_telemetry_central.c)
endif()
